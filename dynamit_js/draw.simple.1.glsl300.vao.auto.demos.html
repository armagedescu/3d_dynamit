API Summary

// Transform matrices
shape.withTransformMatrix4f("matrixName")
shape.transformMatrix4f([...16 floats...])

// Textures  
shape.withTexCoords2d(texCoords)
shape.withTexture(textureId, unit, "samplerName")

// Builders
Builder.polar().formula("1 + 0.5*cos(5*theta)").sectors(60).slices(5).buildConeIndexed()
Builder.cartesian().formula("x*x").domain(-1, 1).sectors(40).buildCylinderIndexedWithColor()

// Translation uniform
shape.withTranslation4f()
shape.translate4f(x, y, z, w)
